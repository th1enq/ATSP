cmake_minimum_required(VERSION 3.20)
project(bottleneck_atsp_thin_tree LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-O3 -DNDEBUG -Wall -Wextra -Wpedantic)
include_directories(${CMAKE_SOURCE_DIR}/include)
add_executable(batsp src/main.cpp)
add_executable(test_core tests/test_core.cpp)
enable_testing()
add_test(NAME core COMMAND test_core)
add_test(NAME certified_witness COMMAND batsp construct-certified ${CMAKE_SOURCE_DIR}/instances/regression/witness6_hk.txt)
add_test(NAME polynomial_baseline_witness COMMAND batsp construct-polynomial-baseline ${CMAKE_SOURCE_DIR}/instances/regression/witness6_hk.txt)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME exact_compression COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_certify_compress.py)
add_test(NAME binary_compression COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/certify_compress.py --kind binary ${CMAKE_SOURCE_DIR}/instances/regression/binary_triangle.txt)
add_test(NAME tree_packing COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_tree_packing.py)
add_test(NAME packing_cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_packing_cli.py)
add_test(NAME weighted_partition_oracle COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_weighted_partition_oracle.py)
add_test(NAME continuation_oracle COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_continuation_oracle.py)
add_test(NAME continuation_state COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_continuation_state.py)
add_test(NAME rational_ellipsoid COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_rational_ellipsoid.py)
add_test(NAME appendix_a_value_oracle COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_appendix_a_value_oracle.py)
